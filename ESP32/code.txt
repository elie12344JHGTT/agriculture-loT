#include <WiFi.h>
#include <PubSubClient.h>

// ====== Configuration WiFi ======
const char* WIFI_SSID     = "iPhone de john";
const char* WIFI_PASSWORD = "cyberbeaugars";

// ====== Configuration MQTT Broker (ton PC Windows qui fait tourner Mosquitto) ======
const char* MQTT_HOST = "broker.hivemq.com";  // IP locale de ton PC (voir note ci-dessous)
const int   MQTT_PORT = 1883;
const char* MQTT_USER = "";
const char* MQTT_PASS = "";
const char* MQTT_CLIENT_ID = "esp32_led_john";

// ====== Topics ======
const char* TOPIC_ACTIONS = "agro-iot/actions";   // Laravel publie ici -> ESP32 s'abonne
// (plus tard pour les capteurs)
const char* TOPIC_SENSORS = "agro-iot/sensors";

// ====== LED ======
const int LED_PIN = 4;  // GPIO 2 = LED bleue embarquée sur la plupart des ESP32

WiFiClient espClient;
PubSubClient client(espClient);

void setup() {
  Serial.begin(115200);
  pinMode(LED_PIN, OUTPUT);
  digitalWrite(LED_PIN, LOW);

  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
  Serial.print("Connexion WiFi...");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nWiFi connecte, IP: " + WiFi.localIP().toString());

  client.setServer(MQTT_HOST, MQTT_PORT);
  client.setCallback(callback);
  client.setBufferSize(256);
}

// Cette fonction est appelée à chaque message reçu sur un topic souscrit
void callback(char* topic, byte* payload, unsigned int length) {
  String message;
  for (unsigned int i = 0; i < length; i++) {
    message += (char)payload[i];
  }
  Serial.print("Message recu sur [");
  Serial.print(topic);
  Serial.print("]: ");
  Serial.println(message);

  // Le message reçu est du JSON : {"actionneur_id":X,"state":"on"} ou "...off"
  if (message.indexOf("\"state\":\"on\"") != -1) {
    digitalWrite(LED_PIN, HIGH);
    Serial.println(">>> LED ALLUMEE");
  } else if (message.indexOf("\"state\":\"off\"") != -1) {
    digitalWrite(LED_PIN, LOW);
    Serial.println(">>> LED ETEINTE");
  }
}

void reconnect() {
  while (!client.connected()) {
    Serial.print("Connexion MQTT au broker...");
    if (client.connect(MQTT_CLIENT_ID, MQTT_USER, MQTT_PASS)) {
      Serial.println("connecte !");
      // S'abonner au topic des commandes actionneurs
      client.subscribe(TOPIC_ACTIONS);
    } else {
      Serial.print("echec, code=");
      Serial.print(client.state());
      Serial.println(" nouvel essai dans 2s");
      delay(2000);
    }
  }
}

void loop() {
  if (!client.connected()) {
    reconnect();
  }
  client.loop();
}
